window.removeBet = function(id) { selected = selected.filter(x => x.id !== id); renderSlip(); }; function renderSlip() { const slipDiv = document.getElementById('slipItems'); if (selected.length === 0) { slipDiv.innerHTML = "Nessuna selezione."; updateCalculation(); return; } slipDiv.innerHTML = selected.map(x => `
${x.match}
${x.pick} (${x.odds})
`).join(''); updateCalculation(); } window.updateCalculation = function() { let totalOdds = selected.reduce((acc, b) => acc * b.odds, 1); let stake = parseFloat(document.getElementById('stake').value) || 0; let potentialWin = totalOdds * stake; document.getElementById('totalOdds').innerText = totalOdds.toFixed(2); document.getElementById('potentialWin').innerText = potentialWin.toFixed(2) + " €"; } loadApp();
